===== Умолчания useradd =====

# useradd -D
GROUP=100
GROUPS=
HOME=/home
INACTIVE=-1
EXPIRE=
SHELL=/bin/sh
SKEL=/etc/skel
USRSKEL=/usr/etc/skel
CREATE_MAIL_SPOOL=no
LOG_INIT=yes

===== Содержимое skel =====

# ls -A /etc/skel                 # локальный шаблон администратора
.bash_logout .bashrc .profile
# ls -A /usr/etc/skel             # шаблон дистрибутива (shadow 4.17)
.bash_logout .ossec_profile

===== Создание пользователя =====

# useradd -m ossec-demo1
# getent passwd ossec-demo1
ossec-demo1:x:1001:1001::/home/ossec-demo1:/bin/sh
# getent group ossec-demo1
ossec-demo1:x:1001:
# stat -c '%A  %U:%G  %n' /home/ossec-demo1
drwxr-x---  ossec-demo1:ossec-demo1  /home/ossec-demo1
# ls -A /home/ossec-demo1         # объединение /etc/skel и /usr/etc/skel
.bash_logout .bashrc .ossec_profile .profile
# getent shadow ossec-demo1
ossec-demo1:!:20774:0:99999:7:::

# grep -E '^(uri|base)' /etc/nslcd.conf

# service nslcd restart >/dev/null 2>&1; sleep 1

===== 5. Переключаем NSS: добавляем 'ldap' к passwd/group/shadow =====

# sed -i -E 's/^(passwd:\s+).*/\1files ldap/' /etc/nsswitch.conf

# sed -i -E 's/^(group:\s+).*/\1files ldap/' /etc/nsswitch.conf

# sed -i -E 's/^(shadow:\s+).*/\1files ldap/' /etc/nsswitch.conf

# grep -E '^(passwd|group|shadow):' /etc/nsswitch.conf
passwd:         files ldap
group:          files ldap
shadow:         files ldap

===== 6. Проверка: ldapuser ОТСУТСТВУЕТ в /etc/passwd... =====

# grep '^ldapuser:' /etc/passwd || echo 'НЕТ в /etc/passwd (files)'
НЕТ в /etc/passwd (files)

===== 7. ...но getent находит его через NSS -> ldap =====

# getent passwd ldapuser
ldapuser:x:5000:5000:LDAP User:/home/ldapuser:/bin/bash

# id ldapuser
uid=5000(ldapuser) gid=5000 groups=5000

Итог: NSS по /etc/nsswitch.conf сначала ищет в files, затем в ldap.
Так один каталог LDAP централизует учётные записи многих машин.

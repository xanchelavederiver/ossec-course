===== Сборка =====

# gcc -Wall -O2 -o pwcheck 11-password-auth.c -lcrypt
# install -m 755 pwcheck /usr/local/bin/pwcheck

===== Верный и неверный пароль =====

# pwcheck bob        # вводится верный пароль bob
Password:
Successfully authenticated: UID=1001

# pwcheck bob        # вводится неверный пароль
Password:
Access denied

# pwcheck mallory    # несуществующее имя
no such user: mallory

===== Без привилегий хеш недоступен =====

$ id
uid=1001(bob) gid=1001(bob) groups=1001(bob),1010(project)
$ pwcheck alice
no permission to read shadow file
$ ls -l /etc/shadow
-rw-r----- 1 root shadow 725 Oct 29 07:25 /etc/shadow

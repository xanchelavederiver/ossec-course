
===== Объекты в Linux =====

$ ls -l /etc/hostname /dev/null /dev/tty
crw-rw-rw- 1 root root 1, 3 Nov 17 18:29 /dev/null
crw-rw-rw- 1 root tty  5, 0 Nov 17 18:29 /dev/tty
-rw-r--r-- 1 root root   10 Nov 17 17:39 /etc/hostname

$ ls -ld /home /proc /sys
drwxr-xr-x   6 root root 4096 Nov 17 18:29 /home
dr-xr-xr-x 128 root root    0 Nov 17 18:29 /proc
dr-xr-xr-x  13 root root    0 Nov 17 18:29 /sys

$ mkfifo /tmp/queue; ls -l /tmp/queue
prw-r--r-- 1 bob bob 0 Nov 17 18:30 /tmp/queue
$ python3 -c 'import socket; socket.socket(socket.AF_UNIX).bind("/tmp/app.sock")'
$ ls -l /tmp/app.sock
srwxr-xr-x 1 bob bob 0 Nov 17 18:30 /tmp/app.sock

$ sleep 30 & ls -ld /proc/$!
dr-xr-xr-x 9 bob bob 0 Nov 17 18:30 /proc/2165
$ grep -E '^(Name|Uid|Gid)' /proc/$!/status
Name:   sleep
Uid:    1001    1001    1001    1001
Gid:    1001    1001    1001    1001


===== Объекты в Linux =====

$ ls -l /etc/hostname /dev/null /dev/tty
crw-rw-rw- 1 root root 1, 3 Oct 19 11:06 /dev/null
crw-rw-rw- 1 root root 5, 0 Oct 19 11:06 /dev/tty
-rw-r--r-- 1 root root   13 Oct 19 11:06 /etc/hostname

$ ls -ld /home /proc /sys
drwxr-xr-x   1 root root 4096 Oct 19 11:06 /home
dr-xr-xr-x 454 root root    0 Oct 19 11:06 /proc
dr-xr-xr-x  13 root root    0 Oct 19 11:07 /sys

$ mkfifo /tmp/queue; ls -l /tmp/queue
prw-r--r-- 1 bob bob 0 Oct 19 11:08 /tmp/queue
$ python3 -c 'import socket; socket.socket(socket.AF_UNIX).bind("/tmp/app.sock")'
$ ls -l /tmp/app.sock
srwxr-xr-x 1 bob bob 0 Oct 19 11:08 /tmp/app.sock

$ sleep 30 & ls -ld /proc/$!
dr-xr-xr-x 9 bob bob 0 Oct 19 11:08 /proc/2567
$ grep -E '^(Name|Uid|Gid)' /proc/$!/status
Name:   sleep
Uid:    1001    1001    1001    1001
Gid:    1001    1001    1001    1001

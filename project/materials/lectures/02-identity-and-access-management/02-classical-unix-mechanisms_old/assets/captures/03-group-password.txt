
===== Создаём группу project и пользователя carol (НЕ входит в project) =====

# groupadd project

# useradd -m -s /bin/bash carol

===== Строки в /etc/group и /etc/gshadow ДО установки пароля =====

# grep '^project:' /etc/group
project:x:1001:

# grep '^project:' /etc/gshadow
project:!::

===== Устанавливаем пароль на группу командой chgpasswd =====

# echo 'project:secret123' | chgpasswd

===== Строки ПОСЛЕ установки пароля (хэш во 2-м поле gshadow) =====

# grep '^project:' /etc/group
project:x:1001:

# grep '^project:' /etc/gshadow
project:$y$j9T$IsLT7wVRqlvJyxI7Np15H1$DilF.GlVBYMFxgTwGOGQ3r2SjCIs.1F8Eq5yv9991C/::

===== carol пытается сменить первичную группу на project =====
# newgrp project  <-- запросит пароль группы, если carol не член группы
Пароль группы позволяет временно войти в группу через newgrp/sg,
не будучи постоянным её членом. Хэш хранится в /etc/gshadow (поле 2).

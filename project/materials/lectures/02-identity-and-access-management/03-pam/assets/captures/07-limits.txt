===== pam_limits: лимит в /etc/security/limits.conf =====

# echo 'bob    hard    fsize    1024' >> /etc/security/limits.conf
# tail -1 /etc/security/limits.conf
bob    hard    fsize    1024
# grep pam_limits /etc/pam.d/sshd
session    required     pam_limits.so

===== Сеанс bob: лимит действует =====

$ ssh bob@127.0.0.1

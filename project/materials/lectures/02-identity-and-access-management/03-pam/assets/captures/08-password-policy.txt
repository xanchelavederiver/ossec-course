===== pam_pwhistory: запрет повторного использования =====

# grep -Ev '^#|^$' /etc/pam.d/common-password | head -3
password        requisite                       pam_pwquality.so retry=7 minlen=10 dcredit=-2 ucredit=-2 lcredit=-2 ocredit=-2 dictcheck=0
password        requisite                       pam_pwhistory.so remember=3 use_authtok
password        [success=1 default=ignore]      pam_unix.so obscure use_authtok try_first_pass yescrypt

===== use_first_pass и try_first_pass =====

# cat /etc/pam.d/demo-g
auth    required        pam_unix.so
auth    required        pam_unix.so
# pamtester demo-g bob authenticate
Password:                          # верный пароль bob – запрошен один раз
pamtester: successfully authenticated

# cat /etc/pam.d/demo-h
auth    required        pam_unix.so use_first_pass
# pamtester demo-h bob authenticate
pamtester: Authentication failure  # пароль не запрошен
# tail -n 1 /var/log/auth.log
19:06:03 srv pamtester: pam_unix(demo-h:auth): auth could not identify password for [bob]

# cat /etc/pam.d/demo-h3
auth    required        pam_unix.so try_first_pass
# pamtester demo-h3 bob authenticate
Password:                          # верный пароль bob
pamtester: successfully authenticated

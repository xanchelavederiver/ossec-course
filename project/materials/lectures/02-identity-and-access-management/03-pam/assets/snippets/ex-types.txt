# auth – подлинность: кто вы?
auth      required    pam_unix.so              # спросить пароль, сверить хеш с /etc/shadow

# account – допуск: можно ли вам сейчас?
account   required    pam_unix.so              # срок действия учётной записи и пароля
account   required    pam_access.so            # access.conf: кому и откуда разрешено

# password – смена аутентификатора
password  required    pam_pwquality.so retry=3 # оценить новый пароль
password  required    pam_unix.so use_authtok  # записать его хеш в /etc/shadow

# session – до и после сеанса
session   required    pam_limits.so            # лимиты ресурсов на входе
session   required    pam_unix.so              # записать открытие и закрытие сеанса

===== Без setuid: эффективный UID равен реальному =====

$ gcc -o userinfo userinfo.c
$ ls -l userinfo
-rwxr-xr-x 1 bob bob 16200 Oct 25 22:32 userinfo
$ ./userinfo
Реальный UID  (кто запустил):      1001 (bob)
Эффективный UID (кем исполняется): 1001 (bob)
$ sudo -u carol ./userinfo
Реальный UID  (кто запустил):      1002 (carol)
Эффективный UID (кем исполняется): 1002 (carol)

===== setuid: эффективный UID равен UID владельца файла =====

$ sudo chown alice:alice userinfo
$ sudo chmod u+s userinfo
$ ls -l userinfo
-rwsr-xr-x 1 alice alice 16200 Oct 25 22:32 userinfo
$ ./userinfo
Реальный UID  (кто запустил):      1001 (bob)
Эффективный UID (кем исполняется): 1003 (alice)
$ sudo -u carol ./userinfo
Реальный UID  (кто запустил):      1002 (carol)
Эффективный UID (кем исполняется): 1003 (alice)

===== Бит s без права x владельца: буква S =====

$ sudo chmod u-x userinfo
$ ls -l userinfo
-rwSr-xr-x 1 alice alice 16200 Oct 25 22:32 userinfo
$ sudo -u alice ./userinfo
sudo: cannot execute '/home/bob/demo03/userinfo': Permission denied (os error 13)
$ ./userinfo
Реальный UID  (кто запустил):      1001 (bob)
Эффективный UID (кем исполняется): 1003 (alice)

===== setuid на скрипте игнорируется =====

$ printf "#!/bin/bash\n\nid\n" > id.sh
$ chmod a+x id.sh
$ ./id.sh
uid=1001(bob) gid=1001(bob) groups=1001(bob),27(sudo),100(users)

$ sudo chown alice id.sh
$ sudo chmod u+s id.sh
$ ls -l id.sh
-rwsr-xr-x 1 alice bob 16 Oct 25 22:32 id.sh
$ ./id.sh
uid=1001(bob) gid=1001(bob) groups=1001(bob),27(sudo),100(users)

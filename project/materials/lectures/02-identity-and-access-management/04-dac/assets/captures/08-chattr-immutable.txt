===== Атрибут i: файл не изменит даже root =====

# chattr +i common-*
# lsattr . | grep common
----i---------e------- ./common-session-noninteractive
----i---------e------- ./common-password
----i---------e------- ./common-session
----i---------e------- ./common-account
----i---------e------- ./common-auth

# cp common-password /tmp/common-password
# echo "# edited again" >> common-password
bash: common-password: Operation not permitted
# rm common-password
rm: cannot remove 'common-password': Operation not permitted
# mv common-password common-password.moved
mv: cannot move 'common-password' to 'common-password.moved': Operation not permitted

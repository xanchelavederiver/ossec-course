
===== Привилегированному процессу нужен бит x хотя бы в одном правиле =====

$ gcc -o 3 hello.c
$ ./3
Hello!
$ chmod ugo-x 3
$ ls -l 3
-rw-r--r-- 1 bob bob 15952 Oct 25 22:35 3
$ sudo ./3
sudo: cannot execute '/home/bob/demo14/3': Permission denied (os error 13)

$ setfacl -m g:users:--x 3
$ getfacl --omit-header 3
user::rw-
group::r--
group:users:--x
mask::r-x
other::r--

$ ./3
bash: ./3: Permission denied
$ sudo ./3
Hello!

===== Права владельца: ACL_USER_OBJ, маска не применяется =====

$ echo "Hello" > 4.txt
$ setfacl -m u:alice:r 4.txt
$ chmod u-r 4.txt
$ ls -l 4.txt
--w-r--r--+ 1 bob bob 6 Oct 25 22:35 4.txt
$ getfacl --omit-header 4.txt
user::-w-
user:alice:r--
group::r--
mask::r--
other::r--

$ cat 4.txt
cat: 4.txt: Permission denied

$ chmod u+r 4.txt
$ cat 4.txt
Hello

$ setfacl -m mask::--x 4.txt
$ getfacl --omit-header 4.txt
user::rw-

group::r--      #effective:---
mask::--x
other::r--

$ cat 4.txt
Hello
$ sudo -u alice cat 4.txt

===== Группа владельцев: ACL_GROUP_OBJ =====

$ echo "Hello" > 6.txt
$ sudo chown carol 6.txt
$ sudo setfacl -m group:nogroup:-wx 6.txt
$ getfacl 6.txt
# file: 6.txt
# owner: carol
# group: bob
user::rw-
group::r--
group:nogroup:-wx
mask::rwx
other::r--


$ id -gn
bob
$ cat 6.txt
Hello

$ sudo -g nogroup groups
nogroup sudo users bob
$ sudo -g nogroup cat 6.txt
Hello

===== Именованная группа: ACL_GROUP =====

$ echo "Hello" > 7.txt
$ sudo chown carol:carol 7.txt
$ sudo chmod 0000 7.txt
$ sudo setfacl -m group:bob:rwx 7.txt
$ getfacl 7.txt
# file: 7.txt
# owner: carol
# group: carol
user::---
group::---
group:bob:rwx
mask::rwx
other::---


$ groups
bob sudo users
$ cat 7.txt
Hello

$ sudo -u nobody -g nogroup groups
nogroup
$ sudo -u nobody -g nogroup cat 7.txt
cat: 7.txt: Permission denied

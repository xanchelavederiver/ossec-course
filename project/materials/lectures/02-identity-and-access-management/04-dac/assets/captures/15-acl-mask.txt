
===== chmod изменяет ACL_GROUP_OBJ минимального списка =====

$ touch file
$ ls -l file
-rw-r--r-- 1 bob bob 0 Oct 25 22:35 file
$ getfacl --omit-header file
user::rw-
group::r--
other::r--


$ chmod 0770 file
$ ls -l file
-rwxrwx--- 1 bob bob 0 Oct 25 22:35 file
$ getfacl --omit-header file
user::rwx
group::rwx
other::---


===== chmod изменяет ACL_MASK расширенного списка =====

$ setfacl -m user:alice:rw file
$ ls -l file
-rwxrwx---+ 1 bob bob 0 Oct 25 22:35 file
$ getfacl --omit-header file
user::rwx
user:alice:rw-
group::rwx
mask::rwx
other::---


$ chmod 0333 file
$ ls -l file
--wx-wx-wx+ 1 bob bob 0 Oct 25 22:35 file
$ getfacl --omit-header file
user::-wx
user:alice:rw-  #effective:-w-
group::rwx      #effective:-wx
mask::-wx
other::-wx


===== Изменение маски отражается на втором триплете вывода ls =====

$ touch file2
$ setfacl -m g:nogroup:w file2
$ ls -l file2
-rw-rw-r--+ 1 bob bob 0 Oct 25 22:35 file2
$ getfacl --omit-header file2
user::rw-
group::r--
group:nogroup:-w-
mask::rw-
other::r--


$ setfacl -m mask::--x file2
$ ls -l file2
-rw---xr--+ 1 bob bob 0 Oct 25 22:35 file2

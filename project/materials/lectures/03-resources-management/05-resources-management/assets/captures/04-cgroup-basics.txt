===== Добавление процесса в группу =====

$ echo $$
4072
$ cat /proc/$$/cgroup
0::/user.slice/user-1001.slice/session-34.scope
$ echo $$ | sudo tee -a /sys/fs/cgroup/demo-cgroup/cgroup.procs
4072
$ cat /sys/fs/cgroup/demo-cgroup/cgroup.procs
4072
4123
$ cat /sys/fs/cgroup/demo-cgroup/pids.current
2
$ cat /proc/$$/cgroup
0::/demo-cgroup
$ cat /sys/fs/cgroup/user.slice/user-1001.slice/session-34.scope/cgroup.procs
4023
4071
